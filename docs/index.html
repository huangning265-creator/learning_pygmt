import gradio as gr import pygmt import numpy as np import matplotlib.pyplot as plt import matplotlib.animation as animation from tempfile import NamedTemporaryFile import os # --- Part 1: PyGMT Earth Relief Plotting Logic --- def plot_earth_relief(projection_type, colormap, show_region): fig = pygmt.Figure() if show_region: region = [-14, 30, 35, 60] grid = pygmt.datasets.load_earth_relief(resolution="10m", region=region) proj_code = "M15c" if projection_type == "Mercator" else "R15c" else: grid = pygmt.datasets.load_earth_relief(resolution="01d") proj_code = "R12c" if projection_type == "Winkel Tripel" else "M12c" fig.grdimage(grid=grid, projection=proj_code, cmap=colormap, frame="a") fig.colorbar(frame=["a2500", "x+lElevation", "y+lm"]) with NamedTemporaryFile(suffix=".png", delete=False) as tmp: fig.savefig(tmp.name) return tmp.name # --- Part 2: 1D Shallow Water Equation Simulation Logic --- def simulate_swe(initial_height, duration): # Parameters nx = 100; dx = 1.0; dt = 0.05 nt = int(duration / dt); g = 9.81 # Initialization x = np.linspace(0, nx*dx, nx) h = np.ones(nx) u = np.zeros(nx) # Initial Condition: Gaussian Bump for i in range(nx): if abs(i - 50) < 10: h[i] = 1.0 + (initial_height - 1.0) * np.exp(-0.1 * (i - 50)**2) h_new = h.copy() u_new = u.copy() # Plot setup fig_sim, (ax1, ax2) = plt.subplots(2, 1, figsize=(8, 6)) plt.subplots_adjust(hspace=0.4) line_h, = ax1.plot(x, h, 'b-', label='Height (h)') ax1.set_ylim(0, initial_height + 1.0) ax1.set_title('1D Shallow Water Equation Simulation') ax1.legend(loc='upper right') ax1.grid(True) line_u, = ax2.plot(x, u, 'r-', label='Velocity (u)') ax2.set_ylim(-2.0, 2.0) ax2.set_xlabel('Distance (m)') ax2.legend(loc='upper right') ax2.grid(True) # Core Numerical Solver (Lax-Friedrichs) def update(frame): nonlocal h, u, h_new, u_new for i in range(1, nx-1): # Mass Conservation (h) h_new[i] = 0.5 * (h[i+1] + h[i-1]) - \ (dt / (2*dx)) * (u[i+1]*h[i+1] - u[i-1]*h[i-1]) # Momentum Conservation (u) - Simplified Non-Conservative Form term1 = 0.5 * (u[i+1] + u[i-1]) term2 = (dt / (2*dx)) * u[i] * (u[i+1] - u[i-1]) term3 = (dt / (2*dx)) * g * (h[i+1] - h[i-1]) u_new[i] = term1 - term2 - term3 # Boundary Conditions (Reflective) h_new[0], h_new[-1] = h_new[1], h_new[-2] u_new[0], u_new[-1] = -u_new[1], -u_new[-2] h[:] = h_new[:] u[:] = u_new[:] line_h.set_ydata(h) line_u.set_ydata(u) return line_h, line_u # Save as GIF output_file = "swe_simulation.gif" ani = animation.FuncAnimation(fig_sim, update, frames=nt, interval=50, blit=False) ani.save(output_file, writer='pillow', fps=20) plt.close(fig_sim) return output_file # --- Part 3: Gradio Interface Setup --- with gr.Blocks(title="Geophysical Teaching Platform") as demo: gr.Markdown("# Geophysical Teaching Platform") gr.Markdown("This platform combines PyGMT Earth Relief visualization and 1D Shallow Water Equation numerical simulation.") with gr.Tab("Earth Relief Plotting (PyGMT)"): gr.Markdown("### Explore PyGMT Plotting Features") with gr.Row(): with gr.Column(): proj_input = gr.Dropdown( choices=["Winkel Tripel", "Mercator"], value="Winkel Tripel", label="Projection" ) cmap_input = gr.Dropdown( choices=["geo", "relief", "turbo", "ocean"], value="geo", label="Color Map" ) region_input = gr.Checkbox( label="Show Specific Region (Europe/N. Africa)", info="Uses region=[-14, 30, 35, 60] with 10m resolution." ) plot_btn = gr.Button("Generate Map", variant="primary") with gr.Column(): map_output = gr.Image(label="PyGMT Output", type="filepath") plot_btn.click( plot_earth_relief, inputs=[proj_input, cmap_input, region_input], outputs=map_output ) with gr.Tab("1D Shallow Water Simulation"): gr.Markdown("### 1D Water Wave Simulation") with gr.Row(): with gr.Column(): height_input = gr.Slider( minimum=1.1, maximum=5.0, value=2.0, label="Initial Wave Height (m)" ) time_input = gr.Slider( minimum=5, maximum=20, value=10, label="Simulation Duration (s)" ) sim_btn = gr.Button("Run Simulation", variant="primary") with gr.Column(): video_output = gr.Image(label="Simulation Result (GIF)") sim_btn.click( simulate_swe, inputs=[height_input, time_input], outputs=video_output ) if __name__ == "__main__": demo.launch()
