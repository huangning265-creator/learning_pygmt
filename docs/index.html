import gradio as gr import pygmt import numpy as np import matplotlib.pyplot as plt import matplotlib.animation as animation from tempfile import NamedTemporaryFile import os # ========================================== # Part 1: PyGMT 地形繪圖邏輯 # ========================================== def plot_earth_relief(projection_type, colormap, show_region): """ 根據使用者選擇的參數繪製地形圖。 資料來源參考提供的 PyGMT 文件。 """ fig = pygmt.Figure() # 1. 載入數據 (Load Earth relief data) # region 參數可用於提供特定區域的數據 if show_region: # 使用文件中提到的特定區域 (歐洲/非洲附近) 和較高解析度 10m # region=[-14, 30, 35, 60] region = [-14, 30, 35, 60] grid = pygmt.datasets.load_earth_relief(resolution="10m", region=region) proj_code = "M15c" if projection_type == "Mercator" else "R15c" else: # 使用文件中提到的全球數據，解析度 01d # resolution="01d" grid = pygmt.datasets.load_earth_relief(resolution="01d") proj_code = "R12c" if projection_type == "Winkel Tripel" else "M12c" # 2. 繪製圖層 (Create a plot) # 使用 grdimage 方法，並依據 cmap 參數設定顏色 fig.grdimage(grid=grid, projection=proj_code, cmap=colormap, frame="a") # 3. 加入色標 (Add a color bar) # 設定 colorbar 的 frame 參數，包含刻度和標籤 fig.colorbar(frame=["a2500", "x+lElevation", "y+lm"]) # 由於 Gradio 需要圖片檔案，我們將 PyGMT 圖形存為暫存檔 with NamedTemporaryFile(suffix=".png", delete=False) as tmp: fig.savefig(tmp.name) return tmp.name # ========================================== # Part 2: 1D 淺水方程式模擬邏輯 (生成動畫) # ========================================== def simulate_swe(initial_height, duration): """ 執行 1D 淺水方程式並回傳 GIF 動畫路徑 """ # 參數設定 nx = 100 dx = 1.0 dt = 0.05 nt = int(duration / dt) g = 9.81 # 初始化 x = np.linspace(0, nx*dx, nx) h = np.ones(nx) u = np.zeros(nx) # 初始條件：高斯波 for i in range(nx): if abs(i - 50) < 10: h[i] = 1.0 + (initial_height - 1.0) * np.exp(-0.1 * (i - 50)**2) h_new = h.copy() u_new = u.copy() # 建立繪圖物件 fig_sim, (ax1, ax2) = plt.subplots(2, 1, figsize=(8, 6)) plt.subplots_adjust(hspace=0.4) line_h, = ax1.plot(x, h, 'b-', label='Height (h)') ax1.set_ylim(0, initial_height + 1.0) ax1.set_ylabel('Water Height (m)') ax1.set_title('1D Shallow Water Equation Simulation') ax1.legend(loc='upper right') ax1.grid(True) line_u, = ax2.plot(x, u, 'r-', label='Velocity (u)') ax2.set_ylim(-2.0, 2.0) ax2.set_xlabel('Distance (m)') ax2.set_ylabel('Velocity (m/s)') ax2.legend(loc='upper right') ax2.grid(True) def update(frame): nonlocal h, u, h_new, u_new # Lax-Friedrichs 方法 for i in range(1, nx-1): h_new[i] = 0.5 * (h[i+1] + h[i-1]) - (dt / (2*dx)) * (u[i+1]*h[i+1] - u[i-1]*h[i-1]) term1 = 0.5 * (u[i+1] + u[i-1]) term2 = (dt / (2*dx)) * u[i] * (u[i+1] - u[i-1]) term3 = (dt / (2*dx)) * g * (h[i+1] - h[i-1]) u_new[i] = term1 - term2 - term3 # 邊界 h_new[0], h_new[-1] = h_new[1], h_new[-2] u_new[0], u_new[-1] = -u_new[1], -u_new[-2] h[:] = h_new[:] u[:] = u_new[:] line_h.set_ydata(h) line_u.set_ydata(u) return line_h, line_u # 儲存為 GIF output_file = "swe_simulation.gif" ani = animation.FuncAnimation(fig_sim, update, frames=nt, interval=50, blit=False) ani.save(output_file, writer='pillow', fps=20) plt.close(fig_sim) return output_file # ========================================== # Part 3: 建構 Gradio 介面 # ========================================== with gr.Blocks(title="地球物理互動教學平台") as demo: gr.Markdown("# 🌊 地球科學與物理模擬教學平台") gr.Markdown("本平台結合了 PyGMT 地形繪圖與淺水方程式數值模擬，讓學生能互動式地探索地球科學概念。") with gr.Tab("🌍 單元一：地球地形繪圖 (PyGMT)"): gr.Markdown("### 探索 PyGMT 的繪圖功能") gr.Markdown("根據 PyGMT 文件，我們可以載入全球或區域地形數據，並設定不同的投影與色標。") with gr.Row(): with gr.Column(): proj_input = gr.Dropdown( choices=["Winkel Tripel", "Mercator"], value="Winkel Tripel", label="投影法 (Projection)" ) cmap_input = gr.Dropdown( choices=["geo", "relief", "turbo", "ocean"], value="geo", label="顏色映射 (Color Map)" ) region_input = gr.Checkbox( label="僅顯示特定區域 (歐洲/北非)", info="勾選後將使用 region=[-14, 30, 35, 60]" ) plot_btn = gr.Button("繪製地圖", variant="primary") with gr.Column(): map_output = gr.Image(label="PyGMT Output", type="filepath") plot_btn.click( plot_earth_relief, inputs=[proj_input, cmap_input, region_input], outputs=map_output ) with gr.Tab("🌊 單元二：淺水方程式模擬 (SWE)"): gr.Markdown("### 1D 淺水波動模擬") gr.Markdown("觀察當我們在水面上製造一個擾動時，波浪與流速如何隨時間變化。") with gr.Row(): with gr.Column(): height_input = gr.Slider( minimum=1.1, maximum=5.0, value=2.0, label="初始波浪高度 (m)" ) time_input = gr.Slider( minimum=5, maximum=20, value=10, label="模擬時長 (秒)" ) sim_btn = gr.Button("開始模擬", variant="primary") with gr.Column(): video_output = gr.Image(label="Simulation Result (GIF)") sim_btn.click( simulate_swe, inputs=[height_input, time_input], outputs=video_output ) if __name__ == "__main__": demo.launch()
